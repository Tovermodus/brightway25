1. Create a database, add an activity, look it up

Registers a new database, creates a node with a production exchange (nothing persists until .save()), then looks it up again with get_node. For loading many activities at once instead of one-by-one, see Database.write() in the bw2data module map.

Code

import bw2data as bd

bd.projects.set_current("my-project")   # isolated project: SQLite DB + data dir

db = bd.Database("example_db")
db.register()                            # required before writing; database now exists, empty

steel = db.new_activity(code="steel", name="steel production", unit="kilogram", location="GLO")
steel.save()                              # nothing persists until .save() is called
steel.new_exchange(input=steel.key, amount=1, type="production").save()

print(len(db))
activity = bd.get_node(database="example_db", name="steel production")
for exc in activity.exchanges():
    print(exc.input, exc.amount, exc["type"])

Output

1
'steel production' (kilogram, GLO, None) 1 production

2. Chimaera vs. non-chimaera nodes

Same steel production inventory, built two ways, to show the difference between a chimaera node and separate process/product nodes. Database.write() auto-classifies each: a self-referencing production exchange (input equal to the node's own key) becomes "processwithreferenceproduct" — a single node holding both the process and its reference product. A production exchange pointing at a different node stays a plain "process", with the product's own data living on a separate "product" node.

Code

import bw2data as bd

bd.projects.set_current("my-project")
db = bd.Database("example_db")
db.register()

db.write({
    # Chimaera: one node IS both the process and its reference product.
    # A self-referencing production exchange (input == this node's own key)
    # makes Database.write() auto-classify it as "processwithreferenceproduct".
    ("example_db", "steel-chimaera"): {
        "name": "steel production",
        "unit": "kilogram",
        "exchanges": [
            {"input": ("example_db", "steel-chimaera"), "amount": 1, "type": "production"},
        ],
    },
    # Non-chimaera: process and product are two separate nodes. The
    # process's production exchange points *away* from itself, at the
    # product node — so it's auto-classified as a plain "process".
    ("example_db", "steel-process"): {
        "name": "steel production",
        "unit": "kilogram",
        "type": "process",
        "exchanges": [
            {"input": ("example_db", "steel-product"), "amount": 1, "type": "production"},
        ],
    },
    ("example_db", "steel-product"): {
        "name": "steel",
        "unit": "kilogram",
        "type": "product",
    },
})

for code in ("steel-chimaera", "steel-process", "steel-product"):
    act = bd.get_node(database="example_db", code=code)
    print(f"{code:16s} type={act['type']:28s} name={act['name']!r}")

Output

steel-chimaera   type=processwithreferenceproduct  name='steel production'
steel-process    type=process                      name='steel production'
steel-product    type=product                      name='steel'

Pick chimaera for simple one-node-per-flow databases (least bookkeeping); pick separate process/product nodes when the reference product needs its own identity — shared by multiple processes, or allocated across co-products via multifunctional. bw2io's strategies.products.separate_processes_from_products() converts chimaera-shaped imports into the separate-node form; see bw2io's note on this.

3. Run a full LCA: biosphere flow + method + bw2calc

Ties bw2data together with bw2calc — a minimal but complete inventory + impact calculation: a process emits CO₂, a Method characterizes it, bw2calc.LCA solves the score.

Code

import bw2data as bd
import bw2calc as bc

bd.projects.set_current("my-project")

biosphere = bd.Database("biosphere")
biosphere.register()
co2 = biosphere.new_node(code="co2", name="carbon dioxide", unit="kilogram", type="emission")
co2.save()

db = bd.Database("example_db")
db.register()
steel = db.new_node(code="steel", name="steel production", unit="kilogram", location="GLO")
steel.save()
steel.new_exchange(input=steel.key, amount=1, type="production").save()
steel.new_exchange(input=co2.key, amount=1.8, type="biosphere").save()   # 1.8 kg CO2 per kg steel

method_key = ("tutorial_method", "gwp")
method = bd.Method(method_key)
method.register(unit="kg CO2-eq")
method.write([(co2.key, 1.0)])            # characterization factor: 1 kg CO2 = 1 kg CO2-eq

lca = bc.LCA({steel.key: 1}, method_key)  # score for producing 1 kg of steel
lca.lci()
lca.lcia()
print(lca.score)

Output

1.7999999523162842

4. Traverse a small supply chain, ranked by cumulative LCA score

Builds a 3-process supply chain (steel ← iron ore + electricity) on top of a solved bw2calc.LCA, then runs NewNodeEachVisitGraphTraversal to find which upstream activities contribute most to the score. node.cumulative_score includes everything further upstream of that node; node.direct_emissions_score is just that node's own biosphere flows.

Code

import bw2data as bd
import bw2calc as bc
from bw_graph_tools import NewNodeEachVisitGraphTraversal, GraphTraversalSettings

bd.projects.set_current("my-project")

biosphere = bd.Database("biosphere")
biosphere.register()
co2 = biosphere.new_node(code="co2", name="carbon dioxide", unit="kilogram", type="emission")
co2.save()

db = bd.Database("example_db")
db.register()

steel = db.new_node(code="steel", name="steel production", unit="kilogram", location="GLO")
steel.save()
iron_ore = db.new_node(code="iron_ore", name="iron ore mining", unit="kilogram", location="GLO")
iron_ore.save()
electricity = db.new_node(code="electricity", name="electricity production", unit="kilowatt hour", location="GLO")
electricity.save()

steel.new_exchange(input=steel.key, amount=1, type="production").save()
steel.new_exchange(input=iron_ore.key, amount=1.6, type="technosphere").save()
steel.new_exchange(input=electricity.key, amount=0.5, type="technosphere").save()
steel.new_exchange(input=co2.key, amount=0.2, type="biosphere").save()

iron_ore.new_exchange(input=iron_ore.key, amount=1, type="production").save()
iron_ore.new_exchange(input=co2.key, amount=0.1, type="biosphere").save()

electricity.new_exchange(input=electricity.key, amount=1, type="production").save()
electricity.new_exchange(input=co2.key, amount=0.9, type="biosphere").save()

method_key = ("tutorial_method", "gwp")
method = bd.Method(method_key)
method.register(unit="kg CO2-eq")
method.write([(co2.key, 1.0)])

lca = bc.LCA({steel.key: 1}, method_key)
lca.lci()
lca.lcia()
print("score:", round(lca.score, 4))

gt = NewNodeEachVisitGraphTraversal(lca, GraphTraversalSettings(cutoff=0.001))
gt.traverse()

for node in sorted(gt.nodes.values(), key=lambda n: n.depth):
    if node.unique_id == -1:
        continue
    act = bd.get_node(id=node.activity_datapackage_id)
    print(f"depth={node.depth} {act['name']:26s} cumulative_score={node.cumulative_score:.4f} direct_score={node.direct_emissions_score:.4f}")

Output

score: 0.81
depth=1 steel production           cumulative_score=0.8100 direct_score=0.2000
depth=2 iron ore mining            cumulative_score=0.1600 direct_score=0.1600
depth=2 electricity production     cumulative_score=0.4500 direct_score=0.4500

5. Find the dominant path between two activities

Same supply chain, but now iron ore mining also consumes electricity directly. path_as_brightway_objects finds the path of greatest multiplicative technosphere flow between two nodes — here it correctly reports that steel's biggest route to electricity is indirect, through iron ore, not the direct edge.

Code

import bw2data as bd
import bw2calc as bc
from bw_graph_tools import path_as_brightway_objects

bd.projects.set_current("my-project")

biosphere = bd.Database("biosphere")
biosphere.register()
co2 = biosphere.new_node(code="co2", name="carbon dioxide", unit="kilogram", type="emission")
co2.save()

db = bd.Database("example_db")
db.register()

steel = db.new_node(code="steel", name="steel production", unit="kilogram", location="GLO")
steel.save()
iron_ore = db.new_node(code="iron_ore", name="iron ore mining", unit="kilogram", location="GLO")
iron_ore.save()
electricity = db.new_node(code="electricity", name="electricity production", unit="kilowatt hour", location="GLO")
electricity.save()

steel.new_exchange(input=steel.key, amount=1, type="production").save()
steel.new_exchange(input=iron_ore.key, amount=1.6, type="technosphere").save()
steel.new_exchange(input=electricity.key, amount=0.5, type="technosphere").save()
steel.new_exchange(input=co2.key, amount=0.2, type="biosphere").save()

iron_ore.new_exchange(input=iron_ore.key, amount=1, type="production").save()
iron_ore.new_exchange(input=electricity.key, amount=3.0, type="technosphere").save()
iron_ore.new_exchange(input=co2.key, amount=0.1, type="biosphere").save()

electricity.new_exchange(input=electricity.key, amount=1, type="production").save()
electricity.new_exchange(input=co2.key, amount=0.9, type="biosphere").save()

method = bd.Method(method_key)
method.register(unit="kg CO2-eq")
method.write([(co2.key, 1.0)])

lca = bc.LCA({steel.key: 1}, method_key)
lca.lci()
lca.lcia()

# Find the path of greatest multiplicative technosphere flow from steel to electricity
# (steel consumes electricity directly, but also indirectly via iron ore, which
# itself consumes a lot of electricity -- path_as_brightway_objects reports which
# route dominates).
path = path_as_brightway_objects(steel, electricity, lca=lca)
for product_node, activity_node, amount in path:
    print(f"{product_node['name']:12s} -> {activity_node['name']:26s} amount={amount:.4f}")

6. Top contributing processes and emissions

ContributionAnalysis's annotated_top_processes/annotated_top_emissions rank a solved LCA's technosphere and biosphere contributions — here methane's much higher characterization factor (28 vs. 1 for CO₂) makes it the top emission despite a tiny mass.

Code

import bw2data as bd
import bw2calc as bc
from bw2analyzer import ContributionAnalysis

bd.projects.set_current("my-project")

biosphere = bd.Database("biosphere")
biosphere.register()
co2 = biosphere.new_node(code="co2", name="carbon dioxide", unit="kilogram", type="emission")
co2.save()
ch4 = biosphere.new_node(code="ch4", name="methane", unit="kilogram", type="emission")
ch4.save()

db = bd.Database("example_db")
db.register()

steel = db.new_node(code="steel", name="steel production", unit="kilogram", location="GLO")
steel.save()
iron_ore = db.new_node(code="iron_ore", name="iron ore mining", unit="kilogram", location="GLO")
iron_ore.save()

steel.new_exchange(input=steel.key, amount=1, type="production").save()
steel.new_exchange(input=iron_ore.key, amount=1.6, type="technosphere").save()
steel.new_exchange(input=co2.key, amount=0.2, type="biosphere").save()
steel.new_exchange(input=ch4.key, amount=0.01, type="biosphere").save()

iron_ore.new_exchange(input=iron_ore.key, amount=1, type="production").save()
iron_ore.new_exchange(input=co2.key, amount=0.1, type="biosphere").save()
iron_ore.new_exchange(input=ch4.key, amount=0.05, type="biosphere").save()

method_key = ("tutorial_method", "gwp")
method = bd.Method(method_key)
method.register(unit="kg CO2-eq")
method.write([(co2.key, 1.0), (ch4.key, 28.0)])   # CH4 has a much higher GWP per kg

lca = bc.LCA({steel.key: 1}, method_key)
lca.lci()
lca.lcia()
print("total score:", round(lca.score, 4))

ca = ContributionAnalysis()

print("\ntop processes:")
for score, supply, activity in ca.annotated_top_processes(lca):
    print(f"  {activity['name']:24s} score={score:.4f} supply={supply:.4f}")

print("\ntop emissions:")
for score, amount, flow in ca.annotated_top_emissions(lca):
    print(f"  {flow['name']:12s} score={score:.4f} amount={amount:.4f}")

9. Build a multi-output process and allocate it by price

"oil refining" has two functional (reference) output edges, diesel and naphtha, each carrying a price property. Database(backend="multifunctional") auto-detects it as type="multifunctional"; writing it (with default_allocation="price" set on the database) runs .allocate(), which splits it into two read-only, single-output ReadOnlyProcessWithReferenceProduct nodes, each carrying its share of the amount.

Code

import bw2data as bd
import multifunctional  # noqa: F401  -- registers the "multifunctional" backend into bw2data

bd.projects.set_current("my-project")

db = bd.Database("example_db", backend="multifunctional")
db.register(default_allocation="price")

# "refining" is a multifunctional process: it has two functional (reference)
# output edges, diesel and naphtha, each carrying a "price" property used
# as the economic allocation factor.
db.write({
    ("example_db", "refining"): {
        "name": "oil refining",
        "location": "GLO",
        "exchanges": [
            {
                "functional": True, "type": "production",
                "name": "diesel", "unit": "kilogram", "amount": 0.6,
                "properties": {"price": 1.2},
            },
            {
                "functional": True, "type": "production",
                "name": "naphtha", "unit": "kilogram", "amount": 0.4,
                "properties": {"price": 0.5},
            },
        ],
    },
})

refining = bd.get_node(database="example_db", code="refining")
print("multifunctional:", refining.multifunctional)

print(f"{refining['code']:10s} type={refining['type']:24s} name={refining['name']!r}")

# Read-only process codes are randomly generated each allocation run
# (mf_allocation_run_uuid), so sort/print by reference product, not code.
for act in sorted(
    (a for a in bd.Database("example_db") if a.get("type") == "readonly_process"),
    key=lambda a: a["reference product"],
):
    print(f"  readonly process, reference product={act['reference product']:8s} production amount={act['production amount']}")

print("\nallocation factors on the parent process's functional exchanges:")
functional_excs = sorted((e for e in refining.exchanges() if e.get("functional")), key=lambda e: e["name"])
for exc in functional_excs:
    print(f"  {exc['name']:8s} amount={exc.amount} price={exc['properties']['price']} allocation_factor={exc['mf_allocation_factor']:.3f}")

Output

  readonly process, reference product=diesel   production amount=0.6

allocation factors on the parent process's functional exchanges:

10. Compare built-in allocation strategies

Re-running .allocate() with a different strategy label replaces the previous read-only processes with a fresh allocation. Same process, three strategies from multifunctional.allocation_strategies: "price" and "mass" (both property-based, normalized by production amount) vs. "equal" (flat 1.0 factor per functional edge, so it just splits by production amount).

Code

import bw2data as bd
import multifunctional  # noqa: F401

bd.projects.set_current("my-project")

db = bd.Database("example_db", backend="multifunctional")
db.register(default_allocation="price")

db.write({
    ("example_db", "refining"): {
        "name": "oil refining",
        "location": "GLO",
        "exchanges": [
            {
                "functional": True, "type": "production",
                "name": "diesel", "unit": "kilogram", "amount": 0.6,
                "properties": {"price": 1.2, "mass": 0.6},
            },
            {
                "functional": True, "type": "production",
                "name": "naphtha", "unit": "kilogram", "amount": 0.4,
                "properties": {"price": 0.5, "mass": 0.4},
            },
        ],
    },
})

refining = bd.get_node(database="example_db", code="refining")

# Re-run allocation with different strategies and compare the resulting
# factors -- switching strategy is just calling .allocate() again with a
# different label; each run replaces the previously generated read-only
# processes (purge_expired_linked_readonly_processes, keyed by a fresh
# mf_allocation_run_uuid stamped on the parent each time).
for strategy in ("price", "mass", "equal"):
    refining.allocate(strategy)
    factors = {
        exc["name"]: exc["mf_allocation_factor"]
        for exc in refining.exchanges() if exc.get("functional")
    }
    print(f"{strategy:6s} -> diesel={factors['diesel']:.3f} naphtha={factors['naphtha']:.3f}")

Output

price  -> diesel=0.783 naphtha=0.217
mass   -> diesel=0.692 naphtha=0.308
equal  -> diesel=0.500 naphtha=0.500

11. Run an LCA against one allocated co-product

The whole point of allocation: after splitting "oil refining" by price, its 3 kg of CO₂ also gets split by the same factor and copied onto each read-only process (rescale_exchange in allocation.py) — so bw2calc.LCA can now solve a normal square matrix for "1 kg of diesel" alone, something it can't do for a multi-output process directly.

Code

import bw2data as bd
import bw2calc as bc
import multifunctional  # noqa: F401

bd.projects.set_current("my-project")

biosphere = bd.Database("biosphere")
biosphere.register()
co2 = biosphere.new_node(code="co2", name="carbon dioxide", unit="kilogram", type="emission")
co2.save()

db = bd.Database("example_db", backend="multifunctional")
db.register(default_allocation="price")

db.write({
    ("example_db", "refining"): {
        "name": "oil refining",
        "location": "GLO",
        "exchanges": [
            {
                "functional": True, "type": "production",
                "name": "diesel", "unit": "kilogram", "amount": 0.6,
                "properties": {"price": 1.2},
            },
            {
                "functional": True, "type": "production",
                "name": "naphtha", "unit": "kilogram", "amount": 0.4,
                "properties": {"price": 0.5},
            },
            # 3 kg CO2 emitted per unit of "refining" activity, split
            # across diesel/naphtha by the same price-based factor.
            {"type": "biosphere", "input": co2.key, "amount": 3.0},
        ],
    },
})

method_key = ("tutorial_method", "gwp")
method = bd.Method(method_key)
method.register(unit="kg CO2-eq")
method.write([(co2.key, 1.0)])

readonly = {a["reference product"]: a for a in bd.Database("example_db") if a.get("type") == "readonly_process"}
diesel, naphtha = readonly["diesel"], readonly["naphtha"]

for act in (diesel, naphtha):
    co2_exc = next(e for e in act.exchanges() if e.input == co2.key)
    print(f"{act['reference product']:8s} co2 exchange amount={co2_exc.amount:.3f}")

    lca = bc.LCA({act.key: 1}, method_key)
    lca.lci()
    lca.lcia()
    print(f"{act['reference product']:8s} LCA score for 1 {act['unit']}: {lca.score:.3f}")

Output

diesel   co2 exchange amount=2.348
diesel   LCA score for 1 kilogram: 3.913
naphtha  co2 exchange amount=0.652
naphtha  LCA score for 1 kilogram: 1.630
